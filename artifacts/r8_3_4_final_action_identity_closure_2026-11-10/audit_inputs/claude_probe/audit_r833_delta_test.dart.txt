              title: 'Circle',
              parts: CirclePlan(parts).toWire(),
              skipped: [for (final e in skipped) e.name],

  group('AUD-R832-1 adversarial — exact Start write', () {
    final day1 = DateTime(2026, 10, 12, 9);
    setUp(() async {
      h = R8Harness(day1);
      await h.boot();
    });

    RecommendationNotifier today(ProviderContainer c) =>
        c.read(recommendationProvider.notifier);
    RecommendationStatus status(ProviderContainer c) =>
        c.read(recommendationProvider).status;

    Future<int> credits(String circleId) async {
      var n = 0;
      for (final c in (await planOnDisk())['cycles'] as List? ?? []) {
        for (final r in (c as Map)['records'] as List) {
          if ((r as Map)['circleId'] == circleId) n++;
        }
      }
      return n;
    }

    int startedInHistory(ProviderContainer c, String circleId) => c
        .read(circleJournalRepositoryProvider)
        .readAll()
        .where((e) => e.circleId == circleId && e.startedAt != null)
        .length;

    Future<void> close(ProviderContainer c) async {
      h.clock.value = h.clock.value.add(const Duration(minutes: 25));
      today(c).close();
      await R8Harness.settle();
    }

    Future<ProviderContainer> restart() async {
      await h.restart();
      final c = h.open();
      c.read(circlePlansProvider);
      c.read(recommendationProvider);
      await R8Harness.settle();
      return c;
    }

    /// Plan paused on disk; offer held; Resume (s1) held; [extra] queues
    /// further lifecycle saves behind it; Start tapped.
    Future<(ProviderContainer, String, HeldWrite, HeldWrite)> setup(
      void Function(ProviderContainer c) extra,
    ) async {
      final c = h.open();
      expect(plans(c).start('getting_going'), PlanActionResult.done);
      await R8Harness.settle();
      final offer = h.disk.hold(recommendationPendingWriteKey);
      today(c).chooseIntention(Intention.moreEnergy, window: TimeWindow.about20);
      await offer.reached;
      final id = c.read(recommendationProvider).recommendation!.circleId;
      expect(plans(c).pause('getting_going'), PlanActionResult.done);
      await R8Harness.settle();
      final s1 = h.disk.hold(circlePlansKey);
      expect(plans(c).resume('getting_going'), PlanActionResult.done);
      await s1.reached;
      extra(c);
      today(c).start();
      await R8Harness.settle();
      expect(status(c), RecommendationStatus.notStarted);
      return (c, id, offer, s1);
    }

    Future<Map<String, Object?>> finish(
      ProviderContainer c,
      String id,
      HeldWrite offer,
    ) async {
      offer.release();
      await R8Harness.settle();
      final outcome = status(c);
      final record = (await planOnDisk())['startedWhilePaused'];
      final starts = startedInHistory(c, id);
      if (outcome == RecommendationStatus.started) await close(c);
      await restart();
      await restart();
      return {
        'outcome': outcome.name,
        'startedWhilePaused': record,
        'historyStarts': starts,
        'credits': await credits(id),
        'lifecycle': (await planOnDisk())['lifecycle'],
      };
    }

    test('P11 s1 Resume REFUSED, s2 Pause and s3 Resume (queued before the '
        'tap) land: the Start was tapped on s3 — the exact save — and may '
        'go ahead', () async {
      final (c, id, offer, s1) = await setup((c) {
        expect(plans(c).pause('getting_going'), PlanActionResult.done);
        expect(plans(c).resume('getting_going'), PlanActionResult.done);
      });
      s1.release(DiskFault.returnsFalse);
      await R8Harness.settle();
      final f = await finish(c, id, offer);
      evidence('P11', f);
      // Either is contract-compliant; started must credit exactly once.
      if (f['outcome'] == 'started') expect(f['credits'], 1);
      if (f['outcome'] == 'notStarted') expect(f['credits'], 0);
    });

    test('P12 s1 Resume refused, no other save: the confirmed Plan is paused '
        '— cancelled, no record, no credit', () async {
      final (c, id, offer, s1) = await setup((_) {});
      s1.release(DiskFault.returnsFalse);
      await R8Harness.settle();
      final f = await finish(c, id, offer);
      evidence('P12', f);
      expect(f['outcome'], 'notStarted');
      expect(f['startedWhilePaused'], isNull);
      expect(f['historyStarts'], 0);
      expect(f['credits'], 0);
    });

    test('P13 s1 Resume lands but its answer is lost (uncertain): cancelled, '
        'no record, no credit', () async {
      final (c, id, offer, s1) = await setup((_) {});
      s1.releaseAckLost();
      await R8Harness.settle();
      final f = await finish(c, id, offer);
      evidence('P13', f);
      expect(f['outcome'], 'notStarted');
      expect(f['historyStarts'], 0);
      expect(f['credits'], 0);
    });

    test('P14 Delete Circle history while the Start is pending', () async {
      final (c, id, offer, s1) = await setup((_) {});
      final deleting = c.read(circleHistoryDeletionProvider).delete();
      deleting.ignore();
      await R8Harness.settle();
      s1.release();
      await R8Harness.settle();
      final f = await finish(c, id, offer);
      final err = await outcome(deleting);
      evidence('P14', {...f, 'deleteError': err?.toString()});
      expect(err, isNull);
      expect(f['startedWhilePaused'], isNull);
      if (f['outcome'] == 'notStarted') expect(f['credits'], 0);
    });

    test('P15 s1 Resume lands (exact save) with a refused unrelated later '
        'save: the Start goes ahead and is credited once', () async {
      final (c, id, offer, s1) = await setup((_) {});
      s1.release();
      await R8Harness.settle();
      final f = await finish(c, id, offer);
      evidence('P15', f);
      expect(f['outcome'], 'started');
      expect(f['credits'], 1);
    });
